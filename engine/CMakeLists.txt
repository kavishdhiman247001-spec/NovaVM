cmake_minimum_required(VERSION 3.20)
project(novavm_probe LANGUAGES CXX)
if(NOT APPLE)
  message(FATAL_ERROR "NovaVM prototype currently targets macOS only.")
endif()
if(NOT CMAKE_SYSTEM_PROCESSOR MATCHES "x86_64|i386")
  message(FATAL_ERROR "This prototype targets Intel x86_64 Macs.")
endif()
add_executable(novavm-engine main.cpp)
target_compile_features(novavm-engine PRIVATE cxx_std_17)
target_compile_options(novavm-engine PRIVATE -Wall -Wextra -Wpedantic)
target_link_libraries(novavm-engine PRIVATE "-framework Hypervisor")
